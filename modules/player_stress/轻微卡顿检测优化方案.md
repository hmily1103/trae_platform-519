# 轻微卡顿检测优化方案

## 🎯 问题描述

**现状**: 人眼可以看到轻微卡顿，但代码中没有提示  
**原因**: 当前检测阈值过于宽松，无法捕获人眼可感知的轻微卡顿  
**目标**: 提升检测敏感度，准确识别人眼可感知的各种程度卡顿  

---

## 📊 当前检测逻辑分析

### 现有阈值问题

```python
# 当前FPS阈值（过于宽松）
FPS < 15  → [严重警告] 
FPS < 20  → [警告]
FPS < 24  → [注意]

# 当前日志卡顿阈值（过于宽松）
log_stutter > 5  → 才报警

# 缺失的检测维度
- 帧间隔一致性检测 ❌
- 瞬时帧率骤降检测 ❌  
- 帧率波动检测 ❌
- 连续性卡顿检测 ❌
```

### 人眼感知特性

1. **帧间隔不均匀**: 即使平均FPS正常，帧间隔不一致也会造成明显卡顿感
2. **瞬时骤降**: 短时间FPS急剧下降，人眼很敏感
3. **持续轻微低帧率**: 20-25fps的持续低帧率人眼可感知
4. **节奏性波动**: FPS在一定范围内波动，造成不流畅感

### 2. 帧间隔一致性检测（核心功能）

```python
import statistics

def detect_frame_interval_inconsistency(self, frame_timestamps):
    """
    检测帧间隔不一致性 - 这是人眼最敏感的卡顿类型
    即使平均FPS正常，但帧间隔不均匀也会造成明显卡顿感
    
    Args:
        frame_timestamps: 帧时间戳列表（毫秒）
    
    Returns:
        tuple: (is_inconsistent, description)
    """
    if len(frame_timestamps) < 10:
        return False, "数据不足"
    
    # 计算帧间隔
    intervals = []
    for i in range(1, len(frame_timestamps)):
        interval = frame_timestamps[i] - frame_timestamps[i-1]
        if interval > 0:  # 过滤异常数据
            intervals.append(interval)
    
    if len(intervals) < 5:
        return False, "有效间隔数据不足"
    
    # 计算间隔的变异系数 (CV = std/mean)
    mean_interval = statistics.mean(intervals)
    std_interval = statistics.stdev(intervals)
    cv = std_interval / mean_interval if mean_interval > 0 else 0
    
    # 人眼感知阈值
    if cv > 0.4:
        return True, f"帧间隔严重不均匀(CV={cv:.2f})，存在明显节奏性卡顿"
    elif cv > 0.3:
        return True, f"帧间隔不均匀(CV={cv:.2f})，存在节奏性卡顿"
    elif cv > 0.2:
        return True, f"帧间隔轻微不均匀(CV={cv:.2f})，可能有微卡顿"
    
    return False, f"帧间隔均匀(CV={cv:.2f})"

def get_frame_timestamps_from_gfxinfo(self, package_name):
    """
    从gfxinfo获取帧时间戳数据
    """
    try:
        gfx_output = self.adb._run_command(
            ["shell", "dumpsys", "gfxinfo", package_name, "framestats"], 
            timeout=3
        )
        
        timestamps = []
        for line in gfx_output.splitlines():
            if line.strip() and not line.startswith('---'):
                parts = line.split(',')
                if len(parts) >= 3:
                    try:
                        # 通常第一列是VSYNC时间戳
                        timestamp_ns = int(parts[0])
                        timestamp_ms = timestamp_ns / 1_000_000
                        timestamps.append(timestamp_ms)
                    except (ValueError, IndexError):
                        continue
        
        return timestamps[-30:] if timestamps else []  # 返回最近30帧
    except Exception:
        return []
```

### 3. 瞬时帧率骤降检测

```python
def detect_fps_drops(self, fps_history):
    """
    检测瞬时帧率骤降 - 短时间内FPS急剧下降
    比如从30fps突然掉到15fps，即使很快恢复也会被人眼察觉
    
    Args:
        fps_history: 最近的FPS历史数据列表
    
    Returns:
        tuple: (has_drop, description)
    """
    if len(fps_history) < 3:
        return False, "FPS历史数据不足"
    
    drops = []
    for i in range(1, len(fps_history)):
        prev_fps = fps_history[i-1]
        curr_fps = fps_history[i]
        
        # 只检测有效FPS值
        if prev_fps > 15 and curr_fps > 0:
            drop_ratio = (prev_fps - curr_fps) / prev_fps
            drop_amount = prev_fps - curr_fps
            
            # 检测显著骤降
            if drop_ratio > 0.4 and drop_amount > 8:  # 骤降40%且超过8fps
                drops.append({
                    'from': prev_fps,
                    'to': curr_fps, 
                    'ratio': drop_ratio,
                    'amount': drop_amount
                })
    
    if drops:
        # 找最严重的骤降
        worst_drop = max(drops, key=lambda x: x['ratio'])
        return True, f"检测到FPS骤降: {worst_drop['from']:.1f}→{worst_drop['to']:.1f} " \
                    f"(降幅{worst_drop['ratio']*100:.0f}%, {worst_drop['amount']:.1f}fps)"
    
    return False, "无明显FPS骤降"
```

### 4. 帧率波动检测

```python
def detect_fps_fluctuation(self, fps_history):
    """
    检测帧率波动 - FPS在一定范围内不稳定波动
    
    Args:
        fps_history: FPS历史数据列表
    
    Returns:
        tuple: (has_fluctuation, description)
    """
    # 过滤有效FPS值
    valid_fps = [fps for fps in fps_history if fps > 0]
    
    if len(valid_fps) < 5:
        return False, "有效FPS数据不足"
    
    fps_mean = statistics.mean(valid_fps)
    fps_std = statistics.stdev(valid_fps)
    fps_min = min(valid_fps)
    fps_max = max(valid_fps)
    fps_range = fps_max - fps_min
    
    # 波动检测阈值
    if fps_std > 6.0:
        return True, f"帧率波动很大(σ={fps_std:.1f}, 范围{fps_min:.1f}-{fps_max:.1f})"
    elif fps_std > 4.0:
        return True, f"帧率波动较大(σ={fps_std:.1f}, 范围{fps_min:.1f}-{fps_max:.1f})"
    elif fps_std > 2.5 and fps_mean < 28:
        return True, f"帧率轻微波动(σ={fps_std:.1f})，且平均帧率偏低({fps_mean:.1f})"
    elif fps_range > 15 and fps_mean < 30:
        return True, f"帧率范围较大({fps_range:.1f}fps)，可能影响流畅度"
    
    return False, f"帧率稳定(σ={fps_std:.1f})"
```

### 5. 连续低帧率检测

```python
def detect_sustained_low_fps(self, fps_history, threshold=25, min_duration=3):
    """
    检测连续低帧率 - 持续一段时间的低帧率
    
    Args:
        fps_history: FPS历史数据
        threshold: 低帧率阈值
        min_duration: 最少连续次数
    
    Returns:
        tuple: (has_sustained_low, description)
    """
    if len(fps_history) < min_duration:
        return False, "数据不足"
    
    consecutive_low = 0
    max_consecutive = 0
    
    for fps in fps_history:
        if fps > 0 and fps < threshold:
            consecutive_low += 1
            max_consecutive = max(max_consecutive, consecutive_low)
        else:
            consecutive_low = 0
    
    if max_consecutive >= min_duration:
        return True, f"检测到连续{max_consecutive}次低帧率(<{threshold}fps)，存在持续性卡顿"
    elif max_consecutive >= 2:
        return True, f"检测到{max_consecutive}次连续低帧率，可能有轻微卡顿"
    
    return False, "无持续低帧率"
```

### 6. 音视频同步检测

```python
def detect_av_sync_issues(self, audio_active, video_fps, audio_underrun_count=0):
    """
    音视频同步问题检测
    
    Args:
        audio_active: 音频是否活跃
        video_fps: 视频帧率
        audio_underrun_count: 音频下溢次数
    
    Returns:
        tuple: (has_sync_issue, description)
    """
    issues = []
    
    # 音频正常但视频帧率异常
    if audio_active and video_fps > 0 and video_fps < 22:
        issues.append(f"音频正常但视频帧率偏低({video_fps:.1f}fps)")
    
    # 音频下溢但视频正常
    if audio_underrun_count > 0 and video_fps > 25:
        issues.append(f"音频下溢{audio_underrun_count}次但视频正常")
    
    # 音视频都有问题
    if audio_underrun_count > 0 and video_fps > 0 and video_fps < 24:
        issues.append(f"音视频都存在问题(音频下溢{audio_underrun_count}次，视频{video_fps:.1f}fps)")
    
    if issues:
        return True, f"可能存在音视频不同步: {'; '.join(issues)}"
    
    return False, "音视频同步正常"
```

### 7. 缓冲区健康度检测

```python
def detect_buffer_health(self, recent_logs):
    """
    缓冲区健康度检测 - 预测性卡顿检测
    在卡顿发生前就能检测到缓冲区压力
    
    Args:
        recent_logs: 最近的日志行列表
    
    Returns:
        tuple: (has_buffer_issue, description)
    """
    buffer_keywords = {
        'critical': ['buffer underrun', 'buffer empty', 'starvation'],
        'warning': ['buffer low', 'buffer warning', 'near underrun', 'buffering'],
        'info': ['rebuffering', 'buffer level', 'buffer fill']
    }
    
    critical_count = 0
    warning_count = 0
    info_count = 0
    
    for log_line in recent_logs:
        log_lower = log_line.lower()
        
        for keyword in buffer_keywords['critical']:
            if keyword in log_lower:
                critical_count += 1
                break
        
        for keyword in buffer_keywords['warning']:
            if keyword in log_lower:
                warning_count += 1
                break
                
        for keyword in buffer_keywords['info']:
            if keyword in log_lower:
                info_count += 1
                break
    
    # 判断缓冲区健康度
    if critical_count > 0:
        return True, f"缓冲区严重不足({critical_count}次严重警告)，即将卡顿"
    elif warning_count > 3:
        return True, f"缓冲区压力较大({warning_count}次警告)，可能即将卡顿"
    elif warning_count > 1 and info_count > 5:
        return True, f"缓冲区状态不稳定(警告{warning_count}次，信息{info_count}次)"
    
    return False, "缓冲区健康"
```

### 8. 人眼感知权重模型

```python
def calculate_perceptual_stutter_score(self, metrics):
    """
    基于人眼感知特性的卡顿评分模型
    不同类型的卡顿对人眼的影响程度不同
    
    Args:
        metrics: 各种检测指标的字典
    
    Returns:
        dict: 包含评分、等级、详情的结果
    """
    score = 0
    details = []
    
    # 权重设计基于人眼敏感度研究
    weights = {
        'frame_interval_inconsistency': 45,  # 最敏感 - 节奏感破坏
        'fps_sudden_drop': 40,              # 很敏感 - 瞬时感知
        'sustained_low_fps': 35,            # 敏感 - 持续感知
        'av_sync_issue': 30,                # 敏感 - 同步感知
        'fps_fluctuation': 25,              # 中等敏感 - 稳定性
        'buffer_pressure': 20,              # 预警性 - 即将发生
        'log_stutter': 15,                  # 技术指标
        'ui_jank': 10                       # 最不敏感 - 可能误报
    }
    
    # 计算各维度得分
    if metrics.get('frame_inconsistent'):
        score += weights['frame_interval_inconsistency']
        details.append(f"帧间隔不均匀({metrics.get('frame_inconsistent_desc', '')})")
    
    if metrics.get('fps_drop_detected'):
        score += weights['fps_sudden_drop']
        details.append(f"瞬时帧率骤降({metrics.get('fps_drop_desc', '')})")
    
    if metrics.get('sustained_low_fps'):
        score += weights['sustained_low_fps']
        details.append(f"持续低帧率({metrics.get('sustained_low_desc', '')})")
    
    if metrics.get('av_sync_issue'):
        score += weights['av_sync_issue']
        details.append(f"音视频同步问题({metrics.get('av_sync_desc', '')})")
    
    if metrics.get('fps_fluctuation'):
        score += weights['fps_fluctuation']
        details.append(f"帧率波动({metrics.get('fps_fluctuation_desc', '')})")
    
    if metrics.get('buffer_issue'):
        score += weights['buffer_pressure']
        details.append(f"缓冲区压力({metrics.get('buffer_desc', '')})")
    
    if metrics.get('log_stutter_count', 0) > 0:
        stutter_score = min(weights['log_stutter'], metrics['log_stutter_count'] * 3)
        score += stutter_score
        details.append(f"日志卡顿({metrics['log_stutter_count']}次)")
    
    if metrics.get('ui_jank_percent', 0) > 15:
        jank_score = min(weights['ui_jank'], (metrics['ui_jank_percent'] - 15) * 2)
        score += jank_score
        details.append(f"UI渲染丢帧({metrics['ui_jank_percent']:.1f}%)")
    
    # 评级系统
    if score >= 80:
        level = "严重卡顿"
        recommendation = "❌ 不建议上线，存在明显卡顿问题"
    elif score >= 50:
        level = "明显卡顿"
        recommendation = "⚠️ 需要优化，用户可明显感知卡顿"
    elif score >= 30:
        level = "轻微卡顿"
        recommendation = "⚠️ 建议优化，部分用户可能感知到卡顿"
    elif score >= 15:
        level = "微卡顿"
        recommendation = "ℹ️ 可接受，但建议关注"
    else:
        level = "流畅"
        recommendation = "✅ 播放流畅，可以上线"
    
    return {
        'score': min(100, score),  # 限制最高100分
        'level': level,
        'details': details,
        'recommendation': recommendation,
        'human_perceptible': score >= 15,  # 15分以上人眼可感知
        'severity': 'high' if score >= 50 else 'medium' if score >= 30 else 'low'
    }
```

### 9. 综合检测主函数

```python
def comprehensive_stutter_detection(self):
    """
    综合卡顿检测 - 整合所有检测维度
    """
    # 收集最近的监控数据
    recent_snapshots = self.history[-20:] if len(self.history) >= 20 else self.history
    if not recent_snapshots:
        return {"level": "unknown", "details": ["数据不足"]}
    
    # 提取数据
    fps_history = [s.get('video_fps', 0) for s in recent_snapshots if s.get('video_fps', 0) > 0]
    current_fps = fps_history[-1] if fps_history else 0
    audio_active = recent_snapshots[-1].get('audio_active', False)
    log_stutter_count = self.log_stutter_count
    
    # 获取帧时间戳数据
    frame_timestamps = self.get_frame_timestamps_from_gfxinfo(self.package_name)
    
    # 执行各项检测
    metrics = {}
    
    # 1. 基础FPS检测
    fps_severity, fps_desc = self.get_fps_severity_level(current_fps)
    metrics['fps_severity'] = fps_severity
    metrics['fps_desc'] = fps_desc
    
    # 2. 帧间隔一致性检测
    frame_inconsistent, frame_desc = self.detect_frame_interval_inconsistency(frame_timestamps)
    metrics['frame_inconsistent'] = frame_inconsistent
    metrics['frame_inconsistent_desc'] = frame_desc
    
    # 3. 瞬时FPS骤降检测
    fps_drop, fps_drop_desc = self.detect_fps_drops(fps_history)
    metrics['fps_drop_detected'] = fps_drop
    metrics['fps_drop_desc'] = fps_drop_desc
    
    # 4. 帧率波动检测
    fps_fluct, fps_fluct_desc = self.detect_fps_fluctuation(fps_history)
    metrics['fps_fluctuation'] = fps_fluct
    metrics['fps_fluctuation_desc'] = fps_fluct_desc
    
    # 5. 连续低帧率检测
    sustained_low, sustained_desc = self.detect_sustained_low_fps(fps_history)
    metrics['sustained_low_fps'] = sustained_low
    metrics['sustained_low_desc'] = sustained_desc
    
    # 6. 音视频同步检测
    av_sync, av_desc = self.detect_av_sync_issues(audio_active, current_fps)
    metrics['av_sync_issue'] = av_sync
    metrics['av_sync_desc'] = av_desc
    
    # 7. 缓冲区健康度检测（需要传入最近的日志）
    # buffer_issue, buffer_desc = self.detect_buffer_health(recent_logs)
    # metrics['buffer_issue'] = buffer_issue
    # metrics['buffer_desc'] = buffer_desc
    
    # 8. 日志卡顿
    metrics['log_stutter_count'] = log_stutter_count
    
    # 9. UI Jank
    recent_jank = sum(s.get('gfx_jank_percent', 0) for s in recent_snapshots[-5:]) / min(5, len(recent_snapshots))
    metrics['ui_jank_percent'] = recent_jank
    
    # 计算人眼感知评分
    perceptual_result = self.calculate_perceptual_stutter_score(metrics)
    
    return perceptual_result
```

- 建议提供用户反馈机制，持续优化阈值

- 过滤异常数据点，避免影响检测准确性
- 建立数据验证机制

## 📚 参考资料

### 人眼视觉感知研究

1. **帧率感知阈值**: 人眼对24fps以下的帧率变化很敏感
2. **节奏感知**: 帧间隔不均匀比平均帧率低更容易被察觉
3. **瞬时感知**: 短时间的帧率骤降（>40%）人眼很容易感知
4. **适应性**: 人眼会适应稳定的低帧率，但对波动很敏感

### Android性能分析

1. **gfxinfo**: 提供详细的帧渲染统计数据
2. **SurfaceFlinger**: 硬件层面的帧率数据
3. **logcat**: 播放器内部的卡顿日志
4. **AudioFlinger**: 音频相关的性能数据
